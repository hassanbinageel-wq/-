//  Classic Technical Analysis MTF — 1H / 4H trend → 1m / 5m entries
//
//  Pure classical technical analysis (no ICT / SMC concepts):
//     • Trend      : EMA 50 / EMA 200 structure on 1H and 4H

//                    2) breakout + retest of a level
//                    3) double bottom / double top (neckline break)
//     • Filters    : RSI (overbought / oversold), volume, trading sessions
//     • Risk       : SL beyond the level / pattern + ATR buffer, TP in R multiples
//
//  Non-repainting: HTF data comes from the previous CLOSED HTF candle and every
//  signal is evaluated on closed bars only. BUY / SELL labels, TP and SL lines

// ───────────────────────────── Helpers ─────────────────────────────
f_key(t) => year(t, TZ) * 10000 + month(t, TZ) * 100 + dayofmonth(t, TZ)
f_px(x)  => str.tostring(x, format.mintick)
f_inSess(s) => not na(time(timeframe.period, s, TZ))
f_tfName(tf) => tf == "60" ? "الساعة" : tf == "240" ? "4 ساعات" : tf == "D" ? "اليومي" : tf
f_ck(b) => b ? "✓ " : "✗ "
f_stars(n) => str.repeat("★", n) + str.repeat("☆", 4 - n)
sigSize = lblSize == Z_H ? size.huge : lblSize == Z_L ? size.large : size.normal

// HTF trend (EMA structure) + last HTF swing levels.
// Shifted by one HTF bar + lookahead_on = previous CLOSED HTF candle only → no repaint.
f_htf(fLen, sLen, pLen) =>
    ef = ta.ema(close, fLen)
    es = ta.ema(close, sLen)
    dir = close > ef and ef > es ? 1 : close < ef and ef < es ? -1 : 0
    hph = ta.pivothigh(high, pLen, pLen)
    hpl = ta.pivotlow(low, pLen, pLen)
    var float lvH = na
    var float lvL = na
    if not na(hph)
        lvH := hph
    if not na(hpl)
        lvL := hpl
    [dir[1], lvH[1], lvL[1]]

[trend1, htfRes, htfSup] = request.security(syminfo.tickerid, htf1, f_htf(emaFast, emaSlow, htfLvlLen), lookahead = barmerge.lookahead_on)
[trend2, _r2, _s2]       = request.security(syminfo.tickerid, htf2, f_htf(emaFast, emaSlow, htfLvlLen), lookahead = barmerge.lookahead_on)

biasLong = switch biasMode
    B_ANY  => trend1 == 1 or trend2 == 1
    B_BOTH => trend1 == 1 and trend2 == 1
    B_HTF2 => trend2 == 1
    B_HTF1 => trend1 == 1
    => true
biasShort = switch biasMode
    B_ANY  => trend1 == -1 or trend2 == -1
    B_BOTH => trend1 == -1 and trend2 == -1
    B_HTF2 => trend2 == -1
    B_HTF1 => trend1 == -1
    => true

tfOK   = timeframe.period == "1" or timeframe.period == "5"
sigOK  = not onlyLTF or tfOK
inSess = not useSess or f_inSess(sLondon) or f_inSess(sNewYork)

finalIdx = finalTP == TP_1 ? 1 : finalTP == TP_2 ? 2 : 3
rrFinal  = finalIdx == 1 ? rr1 : finalIdx == 2 ? rr2 : rr3

dayKey = f_key(time)
var int curDay = na
var int nToday = 0
if dayKey != curDay
    curDay := dayKey
    nToday := 0
canTrade = maxPerDay == 0 or nToday < maxPerDay

// ───────────────────────────── LTF indicators ─────────────────────────────
atr   = ta.atr(14)
rsi   = ta.rsi(close, rsiLen)
emaF  = ta.ema(close, emaFast)
emaS  = ta.ema(close, emaSlow)
volMA = ta.sma(volume, volLen)
volHi = not na(volume) and volume > nz(volMA) and nz(volMA) > 0
zone  = atr * zoneAtr

plot(showEma ? emaF : na, "EMA سريع", color.new(color.orange, 20))
plot(showEma ? emaS : na, "EMA بطيء", color.new(color.purple, 20))

// Candlestick patterns
body   = math.abs(close - open)
rngC   = high - low
upWick = high - math.max(open, close)
dnWick = math.min(open, close) - low
bullEngulf = close > open and close[1] < open[1] and close >= open[1] and open <= close[1]
bearEngulf = close < open and close[1] > open[1] and close <= open[1] and open >= close[1]
hammer     = rngC > 0 and dnWick >= 2 * body and upWick <= body and (close - low) / rngC >= 0.6
shooting   = rngC > 0 and upWick >= 2 * body and dnWick <= body and (high - close) / rngC >= 0.6
bullCandle = bullEngulf or hammer
bearCandle = bearEngulf or shooting

// Support / resistance from swing points
ph = ta.pivothigh(high, lvlLen, lvlLen)
pl = ta.pivotlow(low, lvlLen, lvlLen)
var float res = na
var float sup = na
if not na(ph)
    res := ph
if not na(pl)
    sup := pl
plot(showSR ? res : na, "مقاومة", color.new(colSell, 40), 1, plot.style_linebr)
plot(showSR ? sup : na, "دعم", color.new(colBuy, 40), 1, plot.style_linebr)

// ───────────────────────────── Setups ─────────────────────────────
// 1) Bounce / rejection
bounceL = useBounce and not na(sup) and low <= sup + zone and close > sup and bullCandle
bounceS = useBounce and not na(res) and high >= res - zone and close < res and bearCandle

// 2) Breakout + retest
var float brkUpLvl = na
var int   brkUpBar = na
var float brkDnLvl = na
var int   brkDnBar = na
retestL = useBreak and not na(brkUpLvl) and bar_index > brkUpBar and bar_index - brkUpBar <= retestWin and
     low <= brkUpLvl + zone and close > brkUpLvl and close > open
retestS = useBreak and not na(brkDnLvl) and bar_index > brkDnBar and bar_index - brkDnBar <= retestWin and
     high >= brkDnLvl - zone and close < brkDnLvl and close < open
if not na(brkUpLvl) and (bar_index - brkUpBar > retestWin or close < brkUpLvl - zone)
    brkUpLvl := na
if not na(brkDnLvl) and (bar_index - brkDnBar > retestWin or close > brkDnLvl + zone)
    brkDnLvl := na
if not na(res) and close > res and close[1] <= res and close - open >= brkBody * atr
    brkUpLvl := res
    brkUpBar := bar_index
if not na(sup) and close < sup and close[1] >= sup and open - close >= brkBody * atr
    brkDnLvl := sup
    brkDnBar := bar_index

// 3) Double bottom / double top
var float prevPL   = na
var float phSincePL = na
var float prevPH   = na
var float plSincePH = na
var float dbNeck = na
var float dbLow  = na
var int   dbBar  = na
var float dtNeck = na
var float dtHigh = na
var int   dtBar  = na
if not na(pl)
    if not na(prevPL) and not na(phSincePL) and math.abs(pl - prevPL) <= dblTol * atr
        dbNeck := phSincePL
        dbLow  := math.min(pl, prevPL)
        dbBar  := bar_index
    prevPL    := pl
    phSincePL := na
    plSincePH := na(plSincePH) ? pl : math.min(plSincePH, pl)
if not na(ph)
    if not na(prevPH) and not na(plSincePH) and math.abs(ph - prevPH) <= dblTol * atr
        dtNeck := plSincePH
        dtHigh := math.max(ph, prevPH)
        dtBar  := bar_index
    prevPH    := ph
    plSincePH := na
    phSincePL := na(phSincePL) ? ph : math.max(phSincePL, ph)
doubleL = useDouble and not na(dbNeck) and bar_index - dbBar <= dblWin and close > dbNeck and close[1] <= dbNeck
doubleS = useDouble and not na(dtNeck) and bar_index - dtBar <= dblWin and close < dtNeck and close[1] >= dtNeck
if doubleL or (not na(dbNeck) and (bar_index - dbBar > dblWin or close < dbLow))
    dbNeck := na
if doubleS or (not na(dtNeck) and (bar_index - dtBar > dblWin or close > dtHigh))
    dtNeck := na

// ───────────────────────────── Trade state ─────────────────────────────
var int   tDir    = 0
var int   tBar    = na
var int   tTime   = na
var float tEntry  = na
var float tSL     = na
var float tTP1    = na
var float tFinal  = na
var bool  tBE     = false
var bool  tTP1Hit = false
var int   lastSigTime = na
var int   lastSigDir  = 0

var array<int>   stTime = array.new<int>()
var array<int>   stRes  = array.new<int>()     // 1 win, 0 break-even, -1 loss
var array<float> stR    = array.new<float>()

int    oDir   = 0
float  oSL    = na
int    oScore = 0
string oInfo  = ""

// New signal (closed bars only, one trade at a time)
if tDir == 0 and canTrade and sigOK and inSess and barstate.isconfirmed
    longSet  = bounceL or retestL or doubleL
    shortSet = bounceS or retestS or doubleS
    rsiL = not rsiOn or rsi < rsiOB
    rsiS = not rsiOn or rsi > rsiOS
    emaL = not useEmaLtf or close > emaF
    emaS2 = not useEmaLtf or close < emaF
    d = longSet and biasLong and rsiL and emaL ? 1 : shortSet and biasShort and rsiS and emaS2 ? -1 : 0
    if d != 0
        isB = d == 1 ? bounceL : bounceS
        isR = d == 1 ? retestL : retestS
        setupName = isB ? (d == 1 ? "ارتداد من دعم" : "ارتداد من مقاومة") :
             isR ? "اختراق وإعادة اختبار" : (d == 1 ? "قاع مزدوج" : "قمة مزدوجة")
        lvl = isB ? (d == 1 ? sup : res) : isR ? (d == 1 ? brkUpLvl : brkDnLvl) : (d == 1 ? dbLow : dtHigh)
        stp = d == 1 ? math.min(low, lvl) - slBuf * atr : math.max(high, lvl) + slBuf * atr
        risk = d * (close - stp)
        cTrend = d == 1 ? (trend1 == 1 and trend2 == 1) : (trend1 == -1 and trend2 == -1)
        cHtfLv = (not na(htfSup) and math.abs(lvl - htfSup) <= atr * 1.5) or (not na(htfRes) and math.abs(lvl - htfRes) <= atr * 1.5)
        cCandle = d == 1 ? bullCandle : bearCandle
        cVol = volHi
        score = (cTrend ? 1 : 0) + (cHtfLv ? 1 : 0) + (cCandle ? 1 : 0) + (cVol ? 1 : 0)
        if risk > 0 and risk <= maxSlAtr * atr and score >= minScore
            oDir   := d
            oSL    := stp
            oScore := score
            oInfo  := "\nالنموذج: " + setupName + "\n— عوامل التوافق —\n" + f_ck(cTrend) + "الساعة و4 ساعات بنفس الاتجاه\n" +
                 f_ck(cHtfLv) + "عند مستوى من فريم الساعة\n" + f_ck(cCandle) + "نموذج شمعة قوي\n" + f_ck(cVol) + "فوليوم أعلى من المتوسط" +
                 "\nRSI: " + str.tostring(rsi, "#.0")

bool buySig  = oDir == 1
bool sellSig = oDir == -1
if oDir != 0
    risk    = math.abs(close - oSL)
    tDir    := oDir
    tBar    := bar_index
    tTime   := time
    tEntry  := close
    tSL     := oSL
    tTP1    := close + oDir * risk * rr1
    tFinal  := close + oDir * risk * rrFinal
    tBE     := false
    tTP1Hit := false
    nToday  += 1
    lastSigTime := time
    lastSigDir  := oDir

    tp2 = close + oDir * risk * rr2
    tp3 = close + oDir * risk * rr3
    col = oDir == 1 ? colBuy : colSell
    word = oDir == 1 ? "BUY" : "SELL"
    tip = word + "  |  الدخول " + f_px(close) + "\nالستوب " + f_px(oSL) +
         "\nالهدف 1: " + f_px(tTP1) + " (" + str.tostring(rr1) + "R)" +
         "\nالهدف 2: " + f_px(tp2) + " (" + str.tostring(rr2) + "R)" +
         "\nالهدف 3: " + f_px(tp3) + " (" + str.tostring(rr3) + "R)" + oInfo
    txtSig = word + "\n" + f_px(close) + "\n" + f_stars(oScore)
    if oDir == 1
        label.new(bar_index, low - atr * 0.2, txtSig, style = label.style_label_up,
             color = col, textcolor = color.white, size = sigSize, tooltip = tip)
    else
        label.new(bar_index, high + atr * 0.2, txtSig, style = label.style_label_down,
             color = col, textcolor = color.white, size = sigSize, tooltip = tip)

    x2 = bar_index + lineLen
    if showZones
        box.new(bar_index, close, x2, oSL, border_color = color.new(colSell, 100), bgcolor = color.new(colSell, 85))
        box.new(bar_index, close, x2, tFinal, border_color = color.new(colBuy, 100), bgcolor = color.new(colBuy, 85))
    if showLevels
